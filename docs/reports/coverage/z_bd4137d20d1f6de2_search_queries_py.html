1"""PyMySQL data access for complaint search (FR-017, FR-018). 

2 

3Every user-supplied value is passed as a bound parameter (SEC-003); only fixed SQL fragments are 

4joined together. Functions take an open cursor and never commit. 

5""" 

6from datetime import timedelta 

7 

8from app.tracking.scope import visible_complaints 

9from app.tracking.search_schema import PAGE_SIZE 

10 

11_FROM = ( 

12 "FROM complaints c " 

13 "JOIN categories cat ON cat.category_id = c.category_id " 

14 "LEFT JOIN assignments a ON a.complaint_id = c.complaint_id AND a.is_current = TRUE " 

15 "LEFT JOIN departments d ON d.department_id = a.department_id " 

16 "LEFT JOIN users s ON s.user_id = a.staff_id " 

17)

20def _like(text): 

21 """A LIKE pattern matching `text` anywhere, with LIKE wildcards in the text escaped.""" 

22 escaped = text.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_") 

23 return f"%{escaped}%" 

24 

25 

26def _where(filters, role, user_id): 

27 scope_sql, params = visible_complaints(role, user_id) 

28 conditions = [scope_sql] 

29 

30 if filters["q"]: 

31 # Exact Complaint ID, or a keyword in the title or description. The columns use a 

32 # case-insensitive collation, so LIKE ignores case. 

33 conditions.append("(c.complaint_id = %s OR c.title LIKE %s OR c.description LIKE %s)") 

34 params += [filters["q"], _like(filters["q"]), _like(filters["q"])] 

35 if filters["category_id"]: 

36 conditions.append("c.category_id = %s") 

37 params.append(filters["category_id"]) 

38 if filters["status"]: 

39 conditions.append("c.status = %s") 

40 params.append(filters["status"]) 

41 if filters["department_id"]: 

42 conditions.append("a.department_id = %s") # current assignment only (joined above) 

43 params.append(filters["department_id"]) 

44 if filters["from"]: 

45 conditions.append("c.created_at >= %s") 

46 params.append(filters["from"]) 

47 if filters["to"]: 

48 conditions.append("c.created_at < %s") # inclusive: before the start of the next day 

49 params.append(filters["to"] + timedelta(days=1)) 

50 

51 return " WHERE " + " AND ".join(conditions), params 

52 

53 

54def search(cur, filters, role, user_id): 

55 """(rows for the requested page, total number of matches), newest first.""" 

56 where, params = _where(filters, role, user_id) 

57 

58 cur.execute("SELECT COUNT(*) AS total " + _FROM + where, params) 

59 total = cur.fetchone()["total"] 

60 

61 cur.execute( 

62 "SELECT c.complaint_id, c.title, c.status, c.location, c.created_at, " 

63 "cat.category_id, cat.name AS category_name, " 

64 "a.department_id, d.name AS department_name, a.staff_id, s.name AS staff_name " 

65 + _FROM + where 

66 + " ORDER BY c.created_at DESC, c.complaint_id DESC LIMIT %s OFFSET %s", 

67 params + [PAGE_SIZE, (filters["page"] - 1) * PAGE_SIZE], 

68 ) 

69 return cur.fetchall(), total
